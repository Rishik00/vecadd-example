cmake_minimum_required(VERSION 3.18)
project(VecAdd LANGUAGES CXX CUDA)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CUDA_STANDARD 14)

# Add executable
add_executable(vecadd
    src/main.cpp
    src/host_functions.cpp
    src/kernels/vecadd_kernel.cu
)

target_include_directories(vecadd PRIVATE include)

# Set CUDA architectures (adjust for your GPU)
set_target_properties(vecadd PROPERTIES CUDA_ARCHITECTURES "75;86")
